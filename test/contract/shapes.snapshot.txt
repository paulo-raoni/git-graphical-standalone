    readonly name: string
    readonly node: string

}

type ChooseFolderResponse = import("./result").Failure | import("./result").Success<import("./ipc").ChosenFolder>

interface ChosenFolder {
    readonly path: string
}

interface CommitColumns {
    readonly authorId: Uint32Array<ArrayBufferLike>
    readonly authorTime: Float64Array<ArrayBufferLike>
    readonly authorTzMinutes: Int16Array<ArrayBufferLike>
    readonly commitTime: Float64Array<ArrayBufferLike>
    readonly commitTzMinutes: Int16Array<ArrayBufferLike>
    readonly committerId: Uint32Array<ArrayBufferLike>
    readonly count: number
    readonly hashes: Uint8Array<ArrayBufferLike>
    readonly parentIndices: Uint32Array<ArrayBufferLike>
    readonly parentOffsets: Uint32Array<ArrayBufferLike>
    readonly people: PersonTable
    readonly subjectOffsets: Uint32Array<ArrayBufferLike>
    readonly subjects: string
}

interface Failure {
    readonly exitCode?: number
    readonly kind: "git-missing" | "not-a-repository" | "git-failed" | "cancelled"
    readonly message: string
    readonly ok: false
}

type FailureKind = "git-missing" | "not-a-repository" | "git-failed" | "cancelled"
value FailureKind: { readonly GitMissing: "git-missing"; readonly NotARepository: "not-a-repository"; readonly GitFailed: "git-failed"; readonly Cancelled: "cancelled"; }

interface GitGraphicalBridge {
    readonly applicationInfo: () => Promise<ApplicationInfoResponse>
    readonly chooseFolder: () => Promise<ChooseFolderResponse>
    readonly loadMoreCommits: (request: LoadMoreCommitsRequest) => Promise<LoadMoreCommitsResponse>
    readonly openRepository: (request: OpenRepositoryRequest) => Promise<OpenRepositoryResponse>
    readonly refresh: (request: RefreshRequest) => Promise<RefreshResponse>
}

interface GraphPage {
    readonly commits: CommitColumns
    readonly hasMore: boolean
    readonly layout: LayoutRows
    readonly refs: Refs
    readonly workingTree: WorkingTree
}

    readonly segmentColour: Uint8Array<ArrayBufferLike>
    readonly segmentFromLane: Uint16Array<ArrayBufferLike>

    readonly segmentToLane: Uint16Array<ArrayBufferLike>
}

interface LoadMoreCommitsRequest {
    readonly count: number
}

interface OpenRepositoryRequest {

interface Refs {

    readonly kind: Uint8Array<ArrayBufferLike>
